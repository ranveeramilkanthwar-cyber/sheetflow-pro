Enterprise Cloud Ledger Setup

Connect your Sheet to the Enterprise 4-Tab Relational Architecture with idempotent batch upsert and deletion:

1. Google Sheet Tab 1 (`Transactions`):

Set Row 1 headers in your primary sheet tab:

ID | Date | Type | Category | Bucket | Amount | Mode | Source | Notes
2. Upgraded Apps Script (Batch Sync & Delete with Source Support):

In Google Sheets, go to Extensions ➔ Apps Script, paste this script, and click save: 

function doGet(e) {
  var sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName("Transactions");
  var rows = sheet.getDataRange().getValues();
  if (rows.length <= 1) {
    return ContentService.createTextOutput(JSON.stringify([])).setMimeType(ContentService.MimeType.JSON);
  }
  var headers = rows.shift();
  var data = rows.map(function(r) {
    return {
      id: r[0].toString(),
      date: r[1].toString(),
      type: r[2].toString(),
      category: r[3].toString(),
      bucket: r[4].toString(),
      amount: Number(r[5]),
      mode: r[6].toString(),
      source: r[7] ? r[7].toString() : (r[2] === 'Expense' ? 'Income' : 'Direct'),
      notes: r[8] ? r[8].toString() : ''
    };
  }).reverse();
  
  return ContentService.createTextOutput(JSON.stringify(data))
    .setMimeType(ContentService.MimeType.JSON);
}

function doPost(e) {
  var lock = LockService.getScriptLock();
  lock.tryLock(10000);
  
  try {
    var sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName("Transactions");
    var payload = JSON.parse(e.postData.contents);
    var action = payload.action || "UPSERT";
    var items = payload.items || [payload];
    
    var existingData = sheet.getDataRange().getValues();
    var idMap = {};
    for (var i = 1; i < existingData.length; i++) {
      idMap[existingData[i][0].toString()] = i + 1;
    }

    items.forEach(function(item) {
      var rowIndex = idMap[item.id];
      if (action === "DELETE" && rowIndex) {
        sheet.deleteRow(rowIndex);
      } else if (rowIndex) {
        sheet.getRange(rowIndex, 1, 1, 9).setValues([[
          item.id, item.date, item.type, item.category, item.bucket, item.amount, item.mode, item.source || 'Income', item.notes
        ]]);
      } else if (action !== "DELETE") {
        sheet.appendRow([
          item.id, item.date, item.type, item.category, item.bucket, item.amount, item.mode, item.source || 'Income', item.notes
        ]);
      }
    });

    return ContentService.createTextOutput(JSON.stringify({"status": "synced", "count": items.length}))
      .setMimeType(ContentService.MimeType.JSON);
  } finally {
    lock.releaseLock();
  }
}
